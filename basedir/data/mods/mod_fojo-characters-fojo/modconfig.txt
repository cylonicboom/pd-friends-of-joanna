# Mod: Friends of Joanna

#
# Friends of Joanna ships as four mods:
#   mod_fojo-stages-port4p      setups with only what four players need
#   mod_fojo-stages-fojo        the same plus fojo's own stage changes;
#                               mounted INSTEAD of stages-port4p, never beside it
#   mod_fojo-characters-port4p  characters four players need (empty for now)
#   mod_fojo-characters-fojo    fojo's heads, bodies and bios
#
# Setups are built from pd-fojo/src/setups by tools/mksetups. The default
# build is stages-port4p; -D FOJO_FANFIC adds the #ifdef FOJO_FANFIC hunks
# for stages-fojo.

# this one: mod_fojo-characters-fojo

# Mikado Dark, Poplin Dark, and Calico Dark friends all live here now.
# Mikado's four heads (combat/frock/snow/aqua) are mounted from the JPN
# ROM at runtime; their textures use the per-model naming convention
# (`CheadMikado_XZ/<texid>.bin`) so they don't collide with the PNG
# overrides Catherine/Foslerfer use for the same source texture IDs.

# Mikado Dark (combat) — canonical name for the MP carousel + training bio.
HeadsAndBodies {
    name "head_mikado"
    requiresrom "jpn"
    filenum "CheadMikado_combatZ"
    ismale 0
    yoffset -12
    type 1

# Mikado's bodies: the JPN rom's Cdark_*Z mounted by name, the way her heads
# are. Each differs from the ntsc body in exactly one texture (0x0db2, 0x0db3
# for snow), carried as a per-model texture row in the filetable. Row values
# are the vanilla Joanna rows'. bodyslotnum 1 puts them in the CS body list;
# the spawn chooser picks them by name when Mikado is the operative.
HeadsAndBodies {
    name "body_mikado_combat"
    requiresrom "jpn"
    filenum "CMikado_combatZ"
    handfilenum "GcombathandslodZ"
    ismale 0
    type 1
    height 159
    scale 1.0
    animscale 0.95305162668228
    bodyslotnum 1
    bodyheadname "head_mikado"
