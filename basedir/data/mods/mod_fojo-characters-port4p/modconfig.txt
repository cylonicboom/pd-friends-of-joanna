# Mod: Friends of Joanna
# Author: Catherine Reprobate
#
# Friends of Joanna ships as four mods:
#   mod_fojo-stages-port4p      setups with only what four players need
#   mod_fojo-stages-fojo        the same plus fojo's own stage changes;
#                               mounted INSTEAD of stages-port4p, never beside it
#   mod_fojo-characters-port4p  the heads and bodies, as Combat Simulator characters
#   mod_fojo-characters-fojo    fojo's bios for them
#
# Setups are built from pd-fojo/src/setups by tools/mksetups. The default
# build is stages-port4p; -D FOJO_FANFIC adds the #ifdef FOJO_FANFIC hunks
# for stages-fojo.

# this one: mod_fojo-characters-port4p
#
# Combat Simulator characters. Every head here is also a CS character: a body
# block with bodyslotnum 1, a bodyheadname pairing it with its head, and a
# label, which is the name the CS carousel shows. The internal head/body names
# are unchanged; anything that looks a head up by name finds it here.
# None of these are Team Missions characters.

# ── heads ──────────────────────────────────────────────────────────────

# The jpn rom's Joanna heads (combat/frock/snow/aqua), mounted at runtime and
# named CheadMikado_*Z; their textures use the per-model naming convention
# (`CheadMikado_XZ/<texid>.bin`) so they don't collide with the PNG overrides
# the GE-imported heads use for the same source texture IDs.
HeadsAndBodies {
    name "head_mikado"
    requiresrom "jpn"
    filenum "CheadMikado_combatZ"
    ismale 0
    yoffset -12
    type 1
}
