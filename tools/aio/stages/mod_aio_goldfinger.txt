# DERIVED WORK - NOT ORIGINAL. This mod is a repackaging of
# PD_AIO_March_2026/mod_goldfinger_64 for the pd-fojo modloader. It carries
# the four Goldfinger 64 multiplayer arenas:
#   - the stage declarations below
#   - level data: bgdata/, Ump_setup*Z, P* props
#   - the textures those stages use
#
# Deliberately left out of this mod:
#   - 127 C* character models -> see mod_aio_characters / mod_gex_characters
#   - 38 G* gun and hand models -> byte-identical AIO baseline, not stage content
# Both sets are byte-identical to mod_allinone's copies; every mod in the
# distribution ships the same 165 files.
#
# The source modconfig declares NOTHING - it is comments only. Upstream runs
# these levels by shipping bg files under plain vanilla names and letting
# roster order settle the collision. Every file here has been through slugmod,
# so the names are this mod's alone and the stage blocks below are what makes
# the levels reachable.
#
# The blocks name their levels rather than picking a stage row. A name with no
# stage table row is allocated a free STAGE_EXTRA row and the choice is
# recorded under [MpStageSlots] in pd.ini, so it survives roster changes.
# constants.h labels STAGE_EXTRA20..23 Junkyard, Steel Mill, Mall and Tunnels -
# these four levels, by name - but two of those rows are already claimed by
# mod_aio_stages, so the labels are left as the historical note they are.
#
# Every credit line in the source modconfig.txt is reproduced below
# unaltered. The level work is the original authors'.
# ==========================================================================

# Per-mod model scale overrides, lifted from the canonical AIO source
# 2026-09-20. jonaeru/perfect_dark @ allinone-latest carries a whole
# g_Goldfinger64ModelStates[NUM_MODELS] beside the vanilla table in
# src/game/modeldata/general.c, selected at src/game/propobj.c:2091 on
# g_ModNum == MOD_GOLDFINGER_64. It differs from vanilla in exactly two rows,
# and they are the two props this mod ships its own models for:
#   0x00c1 -> Pmulti_ammo_crate_goldfingerZ
#   0x00e2 -> Pbarrel_goldfingerZ
# Without these the pack's crates and barrels load at the vanilla 1.0 and
# render ten times the size they were authored at. 0x0199 / 4096 is
# 0.099853515625 exactly - the same figure the GE:X block uses.
ModelStates {
  0x00c1 { Scale 0.099853515625 }   # PMULTI_AMMO_CRATE -> Pmulti_ammo_crate_goldfingerZ
  0x00e2 { Scale 0.099853515625 }   # PBARREL           -> Pbarrel_goldfingerZ
}

# NOTE: the four block names below were assigned to the wrong geometry at
# import. AIO's own arena table (jonaeru setup.c g_MpArenas + LmpmenuE) says
# bg_arec = Tunnels, bg_crad = Mall, bg_mp11 = Junkyard, bg_ref = Steel Mill,
# and the arenanames now say so. The block names are kept because they are
# the [MpStageSlots] keys: renaming them would move the levels' rows.

  bgfile "bgdata/bg_arec.seg"
  tilesfile "bgdata/bg_arec_tilesZ"
  padsfile "bgdata/bg_arec_padsZ"
  mpsetupfile "Ump_setuparecZ"
  kind mp
  arenaname "Tunnels"
  # lifted 2026-09-20 (lift-out/lift.py rule 3a): langbank lang.c:240 (STAGE_EXTRA23 -> LANGBANK_AREC); allocation pdmain.c:204 (STAGE_EXTRA23)
  langbank AREC

}
