# --------------------------------------------------------------------------
# NOTE ON STAGE OWNERSHIP - read before adding stages here.
#
# A stage block is what claims a stage for this mod: modConfigParseStage
# writes g_ModStageNums[stagenum] = modnum (port/src/mod.c:1530), and both
# bg geometry texture loading (src/game/bg.c:3050) and the active-mod file
# search (port/src/romdata.c) key off that. On a stage this mod does NOT
# claim, its bgdata and its textures are not consulted at all.
#
# This mod ships level data that belongs to 12 stage(s) by vanilla file
# mapping, of which 0 are claimed by a stage block below. The stage
# blocks below claim 1 stage(s) in total. The 12 unclaimed stages are
# listed here so that the decision to claim them is a deliberate one -
# claiming a stage another mounted mod also claims is a silent
# last-write-wins:
#   0x5  STAGE_EXTRA1
#   0x6  STAGE_EXTRA2
#   0x17  STAGE_MP_RAVINE
#   0x1f  STAGE_MP_COMPLEX
#   0x45  STAGE_MP_VILLA
#   0x4c  STAGE_TEST_MP20
#   0x53  STAGE_EXTRA18
#   0x54  STAGE_EXTRA19
#   0x56  STAGE_EXTRA21
#   0x58  STAGE_EXTRA23
#   0x59  STAGE_EXTRA24
#   0x5b  STAGE_EXTRA26
# --------------------------------------------------------------------------

      0x35 0x36 0x3C 0x46 0x47 0x49 0x55 0x56
      0x5E 0x63 0x69 0x0D 0x16 0x4F 0x4B 0x3F
      0x13 0x19

  }
  # STAGE_24 is in neither g_MpArenas_Vanilla nor g_SoloStages. This block
  # already declares full geometry plus both setups, and Ump_setupmp20Z carries

  # Name taken from the mod's own LmpmenuE string, not invented.
  kind mp
  arenaname "Kakariko Village"
  # lifted 2026-09-20 (lift-out/lift.py rule 3d): langbank lang.c:156 (STAGE_24 -> LANGBANK_SEVX); env env.c:60 (STAGE_24 fog, 'Stormy (Fixed far from 10000 to 20000)')

    water 0 0 0 0x000000
    cloudsheight 0
  }
}

# NO STAGE DECLARATIONS WERE ADDED WITH THE SLUG CONVERSION
#
# Every file in this mod now carries the mod's slug in its name and its path,
# and every manifest entry declares `source: self`, so the loose-file walk no
# longer hands this mod's geometry to a stage row the mod never declared.
#
# All four levels this mod ships - bg_arec, bg_mp13, bg_mp20 and bg_ref - are
# also shipped by mod_gex_stages with different bytes, so each of the four
# primary rows (0x03 STAGE_MP_RAVINE, 0x35 STAGE_MP_VILLA, 0x3c
# STAGE_TEST_MP20, 0x0b STAGE_MP_COMPLEX) is claimed by two packs and the
# conversion declared none of them. They are listed in stage-extra-rows.md at
# the umbrella root, which is where that choice gets made.
#
# The stage 0x24 block above is untouched and still claims Kakariko Village.
# --------------------------------------------------------------------------


# =========================================================================
# Levels this pack ships that had no stage row of their own.
#
# Before the stage space was rowed out these had nowhere to go: the rows
# naming their bg files were already claimed, and claiming one anyway is
# how two packs end up fighting over a level. Each block below names its
# level instead of a row and takes a free STAGE_EXTRA row from the
# allocator, so nothing here contests anything.
#
# ARENA NAMES ARE PLACEHOLDERS in the house form <level>_<pack>, the same
# as cat_gex and crash_site_aio above. Nobody has booted these to see what
# they are. Renaming one is a single string.
#
# Kept at the end of the file on purpose: a bad value in a stage block
# aborts the whole modconfig with no rollback, so a mistake here can only
# cost what is below it.
# =========================================================================
